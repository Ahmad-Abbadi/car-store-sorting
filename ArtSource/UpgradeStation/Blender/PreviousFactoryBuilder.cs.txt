            claw.name="Synty precision tool"; claw.transform.localScale=Vector3.one*.17f;

        private static void CreateDisplayCar(Transform table)
        {
            var prefab=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/ithappy/Modular_Racing_Cars/Prefabs/Car_01/Car_01.prefab");
            var car=Group("Miniature rally car",table);
            car.localRotation=Quaternion.Euler(0,55,0); car.localPosition=new Vector3(0,.035f,0); car.localScale=Vector3.one*.108f;
            var selected=new HashSet<string>{"Car_01","Car_01_Decals","Car_01_Engine","Car_01_Headlight_Simple","Car_01_FrontBumper_Simple","Car_01_RearBumper_Simple","Car_01_Spoiler","Car_01_Wheel_FL","Car_01_Wheel_FR","Car_01_Wheel_RL","Car_01_Wheel_RR"};

            {

                go.AddComponent<MeshFilter>().sharedMesh=filter.sharedMesh;
                go.AddComponent<MeshRenderer>().sharedMaterials=filter.GetComponent<MeshRenderer>().sharedMaterials;
            }
        }

        private static void Strip(GameObject go)
        {
            foreach(var c in go.GetComponentsInChildren<Collider>(true))Object.DestroyImmediate(c);
            foreach(var c in go.GetComponentsInChildren<MonoBehaviour>(true))if(c)Object.DestroyImmediate(c);
        }

        {

        private static Transform Box(Transform parent,string name,Vector3 p,Vector3 size,Material mat,float r)
        {return MeshObject(parent,name,Rounded(size,r),mat,p);}
        private static Mesh Rounded(Vector3 size,float radius)
        {
            var vertices=new List<Vector3>();var normals=new List<Vector3>();var uv=new List<Vector2>();var indices=new List<int>();
            var half=size*.5f;float r=Mathf.Min(radius,Mathf.Min(half.x,Mathf.Min(half.y,half.z))*.98f);var inner=half-Vector3.one*r;
            for(int face=0;face<6;face++)
            {
                int axis=face/2,a=(axis+1)%3,b=(axis+2)%3;float sign=face%2==0?1:-1;int start=vertices.Count;
                float[] xs=r<.015f?new[]{-half[a],-inner[a],inner[a],half[a]}:new[]{-half[a],-inner[a]-r*.7071f,-inner[a],inner[a],inner[a]+r*.7071f,half[a]};
                float[] ys=r<.015f?new[]{-half[b],-inner[b],inner[b],half[b]}:new[]{-half[b],-inner[b]-r*.7071f,-inner[b],inner[b],inner[b]+r*.7071f,half[b]};
                int count=xs.Length;
                for(int y=0;y<count;y++)for(int x=0;x<count;x++)
                {
                    var p=Vector3.zero;p[axis]=half[axis]*sign;p[a]=xs[x];p[b]=ys[y];
                    var c=new Vector3(Mathf.Clamp(p.x,-inner.x,inner.x),Mathf.Clamp(p.y,-inner.y,inner.y),Mathf.Clamp(p.z,-inner.z,inner.z));
                    var n=(p-c).normalized;vertices.Add(c+n*r);normals.Add(n);uv.Add(new Vector2(x/(count-1f),y/(count-1f)));
                }
                for(int y=0;y<count-1;y++)for(int x=0;x<count-1;x++)
                {int i=start+y*count+x;if(sign>0)indices.AddRange(new[]{i,i+1,i+count+1,i,i+count+1,i+count});else indices.AddRange(new[]{i,i+count+1,i+1,i,i+count,i+count+1});}
            }
            var mesh=new Mesh();mesh.SetVertices(vertices);mesh.SetNormals(normals);mesh.SetUVs(0,uv);mesh.SetTriangles(indices,0);mesh.RecalculateBounds();temporary.Add(mesh);return mesh;
        }
        private static Transform Cylinder(Transform parent,string name,Vector3 p,float radius,float height,Material mat,Quaternion rot)
        {
            var primitive=GameObject.CreatePrimitive(PrimitiveType.Cylinder);var mesh=primitive.GetComponent<MeshFilter>().sharedMesh;Object.DestroyImmediate(primitive);
            var t=MeshObject(parent,name,mesh,mat,p);t.localScale=new Vector3(radius*2,height*.5f,radius*2);t.localRotation=rot;return t;
        }
        private static void Rod(Transform p,string name,Vector3 a,Vector3 b,float radius,Material mat)
        {Cylinder(p,name,(a+b)*.5f,radius,(a-b).magnitude,mat,Quaternion.FromToRotation(Vector3.up,b-a));}
        private static void Ring(Transform parent,string name,Vector3 p,float radius,float tube,Material mat)
        {
            var vs=new List<Vector3>();var ns=new List<Vector3>();var ts=new List<int>();
            for(int i=0;i<=48;i++)for(int j=0;j<=8;j++)
            {float a=i*Mathf.PI*2/48,b=j*Mathf.PI*2/8;var n=new Vector3(Mathf.Cos(a)*Mathf.Cos(b),Mathf.Sin(b),Mathf.Sin(a)*Mathf.Cos(b));vs.Add(new Vector3(Mathf.Cos(a)*radius,0,Mathf.Sin(a)*radius)+n*tube);ns.Add(n);}
            for(int i=0;i<48;i++)for(int j=0;j<8;j++){int k=i*9+j;ts.AddRange(new[]{k,k+1,k+10,k,k+10,k+9});}
            var mesh=new Mesh();mesh.SetVertices(vs);mesh.SetNormals(ns);mesh.SetTriangles(ts,0);mesh.RecalculateBounds();temporary.Add(mesh);MeshObject(parent,name,mesh,mat,p);
        }

        private static void Combine(Transform parent,string key)
        {
            var groups=new Dictionary<Material,List<CombineInstance>>();
            foreach(var f in parent.GetComponentsInChildren<MeshFilter>(true))
            {
                var renderer=f.GetComponent<MeshRenderer>();if(!renderer||!f.sharedMesh)continue;
                for(int sub=0;sub<f.sharedMesh.subMeshCount;sub++)
                {
                    var material=renderer.sharedMaterials[Mathf.Min(sub,renderer.sharedMaterials.Length-1)];if(!material)continue;
                    if(!groups.ContainsKey(material))groups[material]=new List<CombineInstance>();
                    groups[material].Add(new CombineInstance{mesh=f.sharedMesh,subMeshIndex=sub,transform=parent.worldToLocalMatrix*f.transform.localToWorldMatrix});
                }
            }
            foreach(var pair in groups)
            {
                string meshName=key+"_"+pair.Key.name.Replace(" ","_");
                var mesh=new Mesh{name=meshName,indexFormat=IndexFormat.UInt32};mesh.CombineMeshes(pair.Value.ToArray(),true,true);mesh.RecalculateBounds();
                string path=Art+"/"+meshName+".asset";
                var existing=AssetDatabase.LoadAssetAtPath<Mesh>(path);
                if(existing){existing.Clear();EditorUtility.CopySerialized(mesh,existing);existing.UploadMeshData(false);EditorUtility.SetDirty(existing);Object.DestroyImmediate(mesh);mesh=existing;}else AssetDatabase.CreateAsset(mesh,path);
                // Output is created after gathering input, then old geometry is removed below.
                MeshObject(parent,"Combined "+pair.Key.name,mesh,pair.Key,Vector3.zero);
            }
            foreach(Transform child in parent.Cast<Transform>().ToArray())if(!child.name.StartsWith("Combined "))Object.DestroyImmediate(child.gameObject);
        }
